# Form Handling in JavaScript

Forms are how users input data on websites. JavaScript lets you read form data, validate it, and handle submissions. This is essential for interactive web pages!

## Getting Form Elements

**By ID:**
```js
let form = document.getElementById("myForm");
```

**By querySelector:**
```js
let form = document.querySelector("#myForm");
let input = document.querySelector("#username");
```

**From form object:**
```js
let form = document.getElementById("myForm");
let username = form.elements.username; // get input with name="username"
```

## Reading Input Values

**Text inputs:**
```js
let input = document.getElementById("username");
let value = input.value;
alert("You entered: " + value);
```

**Checkboxes:**
```js
let checkbox = document.getElementById("agree");
if (checkbox.checked) {
  alert("Checkbox is checked!");
}
```

**Radio buttons:**
```js
let radios = document.getElementsByName("gender");
for (let radio of radios) {
  if (radio.checked) {
    alert("Selected: " + radio.value);
  }
}
```

**Select dropdown:**
```js
let select = document.getElementById("country");
let selectedValue = select.value;
let selectedText = select.options[select.selectedIndex].text;
alert("Selected: " + selectedValue + " (" + selectedText + ")");
```

## Form Events

**Submit event:**
```js
let form = document.getElementById("myForm");

form.addEventListener("submit", function(event) {
  event.preventDefault(); // prevent page reload
  
  // Get form data
  let username = form.elements.username.value;
  let email = form.elements.email.value;
  
  alert("Username: " + username + ", Email: " + email);
});
```

**Input event (fires as user types):**
```js
let input = document.getElementById("username");

input.addEventListener("input", function() {
  alert("You typed: " + input.value);
});
```

**Change event (fires when value changes and loses focus):**
```js
let select = document.getElementById("country");

select.addEventListener("change", function() {
  alert("Country changed to: " + select.value);
});
```

## Form Validation

**Basic validation:**
```js
function validateForm() {
  let username = document.getElementById("username").value;
  let email = document.getElementById("email").value;
  
  // Check if empty
  if (username === "") {
    alert("Username is required!");
    return false;
  }
  
  // Check email format (simple)
  if (!email.includes("@")) {
    alert("Please enter a valid email!");
    return false;
  }
  
  return true; // form is valid
}

let form = document.getElementById("myForm");
form.addEventListener("submit", function(event) {
  if (!validateForm()) {
    event.preventDefault(); // stop submission
  }
});
```

**Show error messages:**
```js
function showError(inputId, message) {
  let input = document.getElementById(inputId);
  input.style.border = "2px solid red";
  
  // Create or update error message
  let errorId = inputId + "Error";
  let errorElement = document.getElementById(errorId);
  
  if (!errorElement) {
    errorElement = document.createElement("div");
    errorElement.id = errorId;
    errorElement.style.color = "red";
    input.parentNode.appendChild(errorElement);
  }
  
  errorElement.textContent = message;
}

function clearError(inputId) {
  let input = document.getElementById(inputId);
  input.style.border = "";
  
  let errorId = inputId + "Error";
  let errorElement = document.getElementById(errorId);
  if (errorElement) {
    errorElement.remove();
  }
}
```

## Accessing All Form Data

**Using FormData object:**
```js
let form = document.getElementById("myForm");

form.addEventListener("submit", function(event) {
  event.preventDefault();
  
  let formData = new FormData(form);
  
  // Get individual values
  let username = formData.get("username");
  let email = formData.get("email");
  
  // Loop through all fields
  for (let [key, value] of formData.entries()) {
    alert(key + ": " + value);
  }
});
```

**Manual collection:**
```js
function getFormData(formId) {
  let form = document.getElementById(formId);
  let data = {};
  
  // Get all inputs
  let inputs = form.querySelectorAll("input, select, textarea");
  
  inputs.forEach(function(input) {
    if (input.type === "checkbox") {
      data[input.name] = input.checked;
    } else if (input.type === "radio") {
      if (input.checked) {
        data[input.name] = input.value;
      }
    } else {
      data[input.name] = input.value;
    }
  });
  
  return data;
}
```

## Setting Form Values

**Set input value:**
```js
document.getElementById("username").value = "Alice";
```

**Check/uncheck checkbox:**
```js
document.getElementById("agree").checked = true;
```

**Select option:**
```js
document.getElementById("country").value = "USA";
```

**Set multiple at once:**
```js
function fillForm(data) {
  for (let key in data) {
    let element = document.querySelector(`[name="${key}"]`);
    if (element) {
      if (element.type === "checkbox") {
        element.checked = data[key];
      } else {
        element.value = data[key];
      }
    }
  }
}

fillForm({
  username: "Alice",
  email: "alice@example.com",
  agree: true
});
```

## Real-Time Validation

**Validate as user types:**
```js
let usernameInput = document.getElementById("username");

usernameInput.addEventListener("input", function() {
  let username = usernameInput.value;
  
  if (username.length < 3) {
    showError("username", "Username must be at least 3 characters");
  } else {
    clearError("username");
  }
});
```

**Validate on blur (when user leaves field):**
```js
let emailInput = document.getElementById("email");

emailInput.addEventListener("blur", function() {
  let email = emailInput.value;
  
  if (!email.includes("@") || !email.includes(".")) {
    showError("email", "Please enter a valid email address");
  } else {
    clearError("email");
  }
});
```

## Complete Form Example

```html
<form id="contactForm">
  <div>
    <label>Name:</label>
    <input type="text" name="name" id="name" required>
    <div id="nameError"></div>
  </div>
  
  <div>
    <label>Email:</label>
    <input type="email" name="email" id="email" required>
    <div id="emailError"></div>
  </div>
  
  <div>
    <label>Message:</label>
    <textarea name="message" id="message" required></textarea>
    <div id="messageError"></div>
  </div>
  
  <button type="submit">Send</button>
</form>
```

```js
let form = document.getElementById("contactForm");

form.addEventListener("submit", function(event) {
  event.preventDefault();
  
  // Get values
  let name = document.getElementById("name").value.trim();
  let email = document.getElementById("email").value.trim();
  let message = document.getElementById("message").value.trim();
  
  // Validate
  let isValid = true;
  
  if (name === "") {
    showError("name", "Name is required");
    isValid = false;
  } else {
    clearError("name");
  }
  
  if (email === "" || !email.includes("@")) {
    showError("email", "Valid email is required");
    isValid = false;
  } else {
    clearError("email");
  }
  
  if (message === "") {
    showError("message", "Message is required");
    isValid = false;
  } else {
    clearError("message");
  }
  
  // If valid, process form
  if (isValid) {
    alert("Form submitted successfully!");
    // Here you would send data to server
    form.reset();
  }
});
```

## Quick Recap

**Getting form elements:**
* `document.getElementById("id")`
* `document.querySelector("#id")`
* `form.elements.name`

**Reading values:**
* `input.value` - text inputs
* `checkbox.checked` - checkboxes
* `select.value` - dropdowns
* `radio.checked` - radio buttons

**Form events:**
* `submit` - when form is submitted
* `input` - as user types
* `change` - when value changes
* `blur` - when field loses focus

**Validation:**
* Check for empty values
* Validate formats (email, etc.)
* Show error messages
* Prevent submission if invalid

**FormData:**
* `new FormData(form)` - easy way to get all data
* `formData.get("name")` - get specific value
* `formData.entries()` - loop through all

**Best practices:**
* Always validate on both client and server
* Show clear error messages
* Validate in real-time for better UX
* Use `preventDefault()` to handle submission yourself
* Reset forms after successful submission

Forms are the bridge between users and your application. Handle them well for a great user experience!
